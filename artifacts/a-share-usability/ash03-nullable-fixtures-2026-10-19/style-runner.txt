import json,subprocess,sys
from pathlib import Path
paths=['tests/api/test_simulated_trading_api_edges.py','tests/component/simulated_trading/test_execution_settlement_uow.py']
checks=[['-m','black','--check',*paths],['-m','isort','--check-only',*paths],['-m','ruff','check',*paths]]
for args in checks:
    print(json.dumps([sys.executable,*args]),flush=True)
    result=subprocess.run([sys.executable,*args],check=False)
    if result.returncode:
        raise SystemExit(result.returncode)
raise SystemExit(subprocess.run(['git','diff','--check','--',*paths],check=False).returncode)
