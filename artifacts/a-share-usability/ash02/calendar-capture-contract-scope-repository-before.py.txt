"""Read calendar capture configuration under the caller's database fence."""

from __future__ import annotations

from django.db import DatabaseError, connections, models, router
from django.db.backends.base.base import BaseDatabaseWrapper

from apps.data_center.application.calendar_capture_configuration import (
    CALENDAR_CAPTURE_CAPABILITY,
    CalendarCaptureBindingCandidate,
    CalendarCaptureConfigurationCandidates,
    CalendarCaptureConfigurationError,
    CalendarCaptureContractCandidate,
    CalendarCapturePolicyCandidate,
    CalendarCaptureProviderCandidate,
)
from apps.data_center.domain.control_plane import CalendarNullableMemberCodecVersion
from apps.data_center.domain.entities import ProviderConfig
from apps.data_center.infrastructure.catalog_models import (
    DatasetContractModel,
    DatasetProviderBindingModel,
    DatasetPublicationPolicyModel,
)
from apps.data_center.infrastructure.models import ProviderConfigModel
from apps.data_center.infrastructure.provider_state_repositories import ProviderConfigRepository
from core.exceptions import ConfigurationError

_DEFAULT_ALIAS = "default"
_POSTGRES_LOCK_TIMEOUT_SECONDS = 5
_CONFIGURATION_MODELS: tuple[type[models.Model], ...] = (
    DatasetContractModel,
    DatasetProviderBindingModel,
    DatasetPublicationPolicyModel,
    ProviderConfigModel,
)


class DjangoCalendarCaptureConfigurationRepository:
    """Read the exact catalog and provider rows without choosing by priority."""

    def __init__(self, provider_configs: ProviderConfigRepository | None = None) -> None:
        """Bind the existing Data Center provider repository as secret owner."""

        self._provider_configs = provider_configs or ProviderConfigRepository()

    @property
    def unit_of_work_key(self) -> str:
        """Return the fixed operational database identity."""

        return "django:default"

    def load_candidates(self, *, using: str) -> CalendarCaptureConfigurationCandidates:
        """Read every explicit calendar candidate while catalog writes are fenced."""

        connection = _require_caller_transaction(using=using)
        _require_model_routes(using=using)
        try:
            _acquire_configuration_fence(connection)
            contracts = tuple(
                CalendarCaptureContractCandidate(
                    contract=row.to_domain(),
                    active=row.active,
                    created_at=row.created_at,
                    updated_at=row.updated_at,
                )
                for row in DatasetContractModel._default_manager.using(using)
                .filter(active=True, calendar_response_max_age_seconds__isnull=False)
                .order_by("dataset_key", "contract_version", "schema_version", "pk")
            )
            policies = tuple(
                CalendarCapturePolicyCandidate(
                    policy=row.to_domain(),
                    active=row.active,
                    created_at=row.created_at,
                    updated_at=row.updated_at,
                )
                for row in DatasetPublicationPolicyModel._default_manager.using(using)
                .filter(
                    active=True,
                    member_codec_version=CalendarNullableMemberCodecVersion.V1.value,
                )
                .order_by("dataset_key", "contract_version", "schema_version", "pk")
            )
            bindings = tuple(
                CalendarCaptureBindingCandidate(
                    binding=row.to_domain(),
                    enabled=row.enabled,
                    created_at=row.created_at,
                    updated_at=row.updated_at,
                )
                for row in DatasetProviderBindingModel._default_manager.using(using)
                .filter(enabled=True, capability=CALENDAR_CAPTURE_CAPABILITY)
                .order_by("dataset_key", "contract_version", "schema_version", "pk")
            )
            providers = tuple(
                CalendarCaptureProviderCandidate(
                    provider_id=int(row.pk),
                    source_type=row.source_type,
                    is_active=row.is_active,
                    created_at=row.created_at,
                    updated_at=row.updated_at,
                )
                for row in ProviderConfigModel._default_manager.using(using)
                .filter(is_active=True, source_type="tushare")
                .order_by("pk")
            )
        except CalendarCaptureConfigurationError:
            raise
        except (AttributeError, DatabaseError, TypeError, ValueError) as error:
            raise _blocked("CALENDAR_CAPTURE_CATALOG_UNAVAILABLE") from error
        return CalendarCaptureConfigurationCandidates(
            contracts=contracts,
            policies=policies,
            bindings=bindings,
            providers=providers,
        )

    def get_provider_config(self, provider_id: int, *, using: str) -> ProviderConfig | None:
        """Resolve one selected provider's secret snapshot through its owner."""

        _require_caller_transaction(using=using)
        _require_model_routes(using=using)
        if type(provider_id) is not int or provider_id < 1:
            raise _blocked("CALENDAR_CAPTURE_PROVIDER_ID_INVALID")
        try:
            return self._provider_configs.get_by_id(provider_id)
        except (ConfigurationError, DatabaseError, RuntimeError, TypeError, ValueError) as error:
            raise _blocked("CALENDAR_CAPTURE_PROVIDER_CONFIGURATION_UNAVAILABLE") from error


def _require_caller_transaction(*, using: str) -> BaseDatabaseWrapper:
    """Require the exact default alias inside an already-owned transaction."""

    if type(using) is not str or using != _DEFAULT_ALIAS:
        raise _blocked("CALENDAR_CAPTURE_DATABASE_ALIAS_UNSUPPORTED")
    connection = connections[using]
    if (
        connection.alias != _DEFAULT_ALIAS
        or not connection.in_atomic_block
        or connection.get_autocommit()
        or not connection.atomic_blocks
    ):
        raise _blocked("CALENDAR_CAPTURE_CALLER_TRANSACTION_REQUIRED")
    return connection


def _require_model_routes(*, using: str) -> None:
    """Reject routers that split any configuration row from the raw store."""

    for model in _CONFIGURATION_MODELS:
        read_alias = router.db_for_read(model) or _DEFAULT_ALIAS
        write_alias = router.db_for_write(model) or _DEFAULT_ALIAS
        if read_alias != using or write_alias != using:
            raise _blocked("CALENDAR_CAPTURE_DATABASE_ALIAS_MISMATCH")


def _acquire_configuration_fence(connection: BaseDatabaseWrapper) -> None:
    """Block catalog/provider absence phantoms until the caller commits."""

    if connection.vendor == "postgresql":
        try:
            with connection.cursor() as cursor:
                cursor.execute("SHOW transaction_isolation")
                isolation = cursor.fetchone()
                if isolation is None or isolation[0] != "read committed":
                    raise _blocked("CALENDAR_CAPTURE_TRANSACTION_ISOLATION_UNSUPPORTED")
                cursor.execute(
                    "SELECT current_setting('lock_timeout')::interval = interval '0' "
                    "OR current_setting('lock_timeout')::interval > interval '5 seconds'"
                )
                should_cap_timeout = cursor.fetchone()
                if should_cap_timeout is None or len(should_cap_timeout) != 1:
                    raise _blocked("CALENDAR_CAPTURE_DATABASE_FENCE_UNAVAILABLE")
                if should_cap_timeout[0]:
                    cursor.execute("SET LOCAL lock_timeout = '5s'")
                table_names = sorted(model._meta.db_table for model in _CONFIGURATION_MODELS)
                for table_name in table_names:
                    cursor.execute(
                        f"LOCK TABLE {connection.ops.quote_name(table_name)} IN SHARE MODE"
                    )
            return
        except CalendarCaptureConfigurationError:
            raise
        except DatabaseError as error:
            raise _blocked("CALENDAR_CAPTURE_DATABASE_FENCE_UNAVAILABLE") from error
    if connection.vendor == "sqlite":
        table_name = connection.ops.quote_name(DatasetContractModel._meta.db_table)
        try:
            with connection.cursor() as cursor:
                cursor.execute(f"UPDATE {table_name} SET dataset_key=dataset_key WHERE 1=0")
        except DatabaseError as error:
            raise _blocked("CALENDAR_CAPTURE_DATABASE_FENCE_UNAVAILABLE") from error
        return
    raise _blocked("CALENDAR_CAPTURE_DATABASE_VENDOR_UNSUPPORTED")


def _blocked(code: str) -> CalendarCaptureConfigurationError:
    """Create a stable operator-facing error without including secret values."""

    return CalendarCaptureConfigurationError(
        "Exchange calendar capture configuration is unavailable or not safely isolated",
        code=code,
    )


__all__ = ["DjangoCalendarCaptureConfigurationRepository"]
