def _runtime_environment() -> str:
    """Mirror the production composition's profile environment selection."""

    module = str(os.environ.get("DJANGO_SETTINGS_MODULE") or "").strip()
    return "production" if module.endswith(".production") else "development"


def _save_instrument_snapshot(
    *, repository: RawLandingRepository, replay_client: object
) -> InstrumentReferenceSnapshotRecord:
    """Parse actual stock_basic rows through RawAudit, mapping, and snapshot ports."""

    from apps.data_center.domain.instrument_reference_source import (
        InstrumentReferenceProviderContract,
    )
    from apps.data_center.infrastructure.instrument_reference_snapshot_repository import (
        DjangoInstrumentReferenceSnapshotRepository,
    )

    crosswalk = InstrumentReferenceExchangeSuffixCrosswalk(
        crosswalk_id="ash02-local-sse-suffix-map",
        crosswalk_version="v1",
        exchange_suffix_pairs=(("SSE", "SH"),),
    )
    contract = InstrumentReferenceProviderContract(
        contract_id="ash02-actual-stock-basic",
        contract_version="v1",
        provider_name="tushare",
        dataset_key="tushare.stock_basic",
        api_name="stock_basic",
        fields=_STOCK_BASIC_FIELDS,
        exchange_scope=("SSE",),
        list_status_scope=("L",),
    )
    candidate = TushareInstrumentReferenceSource(
        replay_client,
        repository,
        DjangoInstrumentReferenceRawAuditOccurrenceWriter(RawAuditRepository()),
        provider_contract=contract,
        exchange_suffix_crosswalk=crosswalk,
    ).fetch_candidate(run_id=uuid4(), ingested_run_id=uuid4())
    source_row = next(row for row in candidate.records if row.ts_code == _ASSET_SOURCE_CODE)
    assert source_row.source_exchange == "SSE"
    assert source_row.source_market == "主板"
    assert source_row.source_list_status == "L"
    assert source_row.list_date == date(1999, 11, 10)
    assert source_row.delist_date is None
    assert source_row.source_observed_at is None
    assert source_row.source_available_at is None

    mapping_content = InstrumentReferenceMappingContent(
        mapping_id="ash02-local-identity-map",
        mapping_version="v1",
        source_crosswalk=crosswalk,
        exchange_rule_market_pairs=(("SSE", "SH"),),
        exchange_source_market_board_pairs=(("SSE", "主板", "MAIN"),),
    )
    snapshot_repository = DjangoInstrumentReferenceSnapshotRepository(
        RawAuditRepository(), repository
    )
    mapping = SaveInstrumentReferenceMappingUseCase(snapshot_repository).execute(mapping_content)
    run_id = candidate.occurrences[0].raw_audit_reference.run_id
    snapshot_content = InstrumentReferenceSnapshotContent(
        run_id=UUID(run_id),
        mapping_record=mapping,
        fetch_rounds=(InstrumentReferenceFetchRound(fetch_round_id=uuid4(), candidate=candidate),),
    )
    saved = SaveInstrumentReferenceSnapshotUseCase(snapshot_repository).execute(snapshot_content)
    reread = GetInstrumentReferenceSnapshotUseCase(snapshot_repository).execute(saved.snapshot_id)
    assert reread == saved
    assert saved.content.record_count == len(candidate.records)
    assert saved.content.source_observed_at is None
    assert saved.content.source_available_at is None
    assert saved.content.must_not_use_for_decision is True
    assert saved.content.must_not_execute is True
    classification = saved.content.classify_security(_ASSET_SOURCE_CODE)
    assert classification is not None
    assert classification.rule_market == "SH"
    assert classification.board == "MAIN"
    assert classification.blocked_reason is None
    assert RawAuditModel.objects.filter(capability="instrument_reference").count() == 1
    return saved


def _create_calendar_catalog() -> tuple[DatasetContractModel, DatasetPublicationPolicyModel]:
    """Create one explicit, test-local calendar contract and a metadata-only provider row."""

    assert not DatasetContractModel.objects.filter(
        active=True, calendar_response_max_age_seconds__isnull=False
    ).exists(), "run only in a dedicated S4 database with no preloaded calendar contract"
    assert not ProviderConfigModel.objects.filter(
        source_type="tushare", is_active=True
    ).exists(), "run only in a dedicated S4 database with no active Tushare provider"
    dataset_key = "tushare.trade_cal"
    contract_version = "ash02-local-validation-v1"
    schema_version = "full-date-calendar-v1"
    contract = DatasetContractModel.objects.create(
        dataset_key=dataset_key,
        contract_version=contract_version,
        schema_version=schema_version,
        owner="ash02-local-validation",
        frequency="reference",
        decision_critical=True,
        fields=[
            {
                "name": "sessions",
                "value_type": "json",
                "nullable": False,
                "zero_allowed": False,
            }
        ],
        calendar_response_max_age_seconds=_SCENARIO_TTL_SECONDS,
        active=True,
    )
    policy = DatasetPublicationPolicyModel.objects.create(
        dataset_key=dataset_key,
        contract_version=contract_version,
        schema_version=schema_version,
        policy_version="ash02-local-calendar-policy-v1",
        member_codec_version=CalendarNullableMemberCodecVersion.V1.value,
        minimum_coverage_ratio=1.0,
        allow_partial=False,
        conflict_action="block",
        required_evidence=["source", "fact_content_hash"],
        retention_days=30,
        active=True,
    )
    DatasetProviderBindingModel.objects.create(
        dataset_key=dataset_key,
        contract_version=contract_version,
        schema_version=schema_version,
        provider="tushare",
        capability=CALENDAR_CAPTURE_CAPABILITY,
        priority=1,
        freshness_seconds=None,
        validator_key=CALENDAR_CAPTURE_VALIDATOR_KEY,
        enabled=True,
    )
    ProviderConfigModel.objects.create(
        name=f"ash02-local-calendar-{uuid4().hex}",
        source_type="tushare",
        is_active=True,
        priority=1,
        description="test-local calendar metadata only; no credentials or network calls",
    )
    return contract, policy


def _save_calendar_publication(
    *,
    repository: RawLandingRepository,
    replay_client: object,
    contract: DatasetContractModel,
    policy: DatasetPublicationPolicyModel,
) -> ExchangeCalendarSnapshot:
    """Publish the exact captured calendar using the production audit and authority path."""

    snapshot_candidate = TushareExchangeCalendarSource(
        replay_client,
        repository,
        dataset_key="tushare.trade_cal",
    ).fetch_candidate("SSE", _CAPTURE_START, _CAPTURE_END)
    assert snapshot_candidate.source_published_at is None
    snapshot = DjangoExchangeCalendarSnapshotRepository().save(snapshot_candidate)
    dataset = DatasetKey(
        value=contract.dataset_key,
        contract_version=contract.contract_version,
        schema_version=contract.schema_version,
    )
    fact = ExchangeCalendarSnapshotFactModel.objects.get(snapshot_id=UUID(snapshot.snapshot_id))
    member = PublicationMember(
        member_id=str(uuid4()),
        publication_id=str(uuid4()),
        dataset_key=dataset.value,
        natural_key=f"SSE:{_CAPTURE_START.isoformat()}:{_CAPTURE_END.isoformat()}",
        source=snapshot.provider_name,
        source_record_id=snapshot.snapshot_id,
        fact_table=ExchangeCalendarSnapshotFactModel._meta.db_table,
        fact_pk=snapshot.snapshot_id,
        observed_at=None,
        raw_payload_hash=snapshot.raw_reference.payload_hash,
        quality_status="accepted",
        revision_number=1,
        available_at=None,
        fetched_at=snapshot.first_seen_at,
        source_published_at=None,
        raw_payload_scope="record_response_body",
        fact_content_hash=canonical_fact_content_hash(fact),
    )
    member_reference = CalendarPublicationMemberReference(
        member=member,
        snapshot_content_hash=snapshot.content_hash,
        snapshot_record_digest=snapshot.record_digest,
        codec_version=CalendarNullableMemberCodecVersion.V1,
    )
    seal = create_calendar_formal_publication_seal(
        dataset=dataset,
        exchange="SSE",
        calendar_response_max_age_seconds=_SCENARIO_TTL_SECONDS,
        contract_recorded_at=contract.created_at,
        policy=policy.to_domain(),
        policy_recorded_at=policy.created_at,
        snapshot=snapshot,
        member_reference=member_reference,
    )
    request = CalendarFormalPublicationRequest(
        proposed_seal=seal,
        expected_pointer=CalendarPublicationPointerExpectation(
            dataset_key=dataset.value,
            publication_key=seal.publication_key,
            pointer_id=None,
            publication_id=None,
            publication_hash=None,
            activation_id=None,
            updated_at=None,
        ),
        activation_id=f"ash02-{uuid4().hex}",
    )
    _publisher_user, proof = _actual_publisher(f"ash02-calendar-publisher-{uuid4().hex}")
    authority = DjangoCalendarPublisherAuthorityRepository(using="default")
    publication_repository = DjangoFormalExchangeCalendarPublicationRepository(
        audit_port=_real_audit_bridge(),
        using="default",
    )
    published = _authorized_publish(
        authority=authority,
        use_case=PublishExchangeCalendarSnapshotUseCase(publication_repository),
        proof=proof,
        request=request,
    )
    assert published.record.publication_id
    return snapshot


def _import_local_rule_and_fee(*, security_id: str, authored_at: datetime) -> None:
    """Persist explicit, test-owned Risk V3 and Fee V2 terms with real repositories."""

    valid_until = _MONDAY_SCENARIO + timedelta(days=2)
    observed_at = authored_at - timedelta(seconds=1)
    rule_v1 = AShareExecutionRule(
        rule_id=f"ash02-local-rule-{uuid4().hex}",
        rule_version="v1",
        security_id=security_id,
        market="SH",
        board="MAIN",
        buy_quantity=ExecutionQuantityConstraint(
            minimum=Decimal("100"), step=Decimal("100"), maximum=Decimal("100000")
        ),
        sell_quantity=ExecutionQuantityConstraint(
            minimum=Decimal("100"), step=Decimal("100"), maximum=Decimal("100000")
        ),
        sell_odd_lot_mode=SellOddLotMode.FORBID,
        price_tick=Decimal("0.01"),
        rule_source_id="ash02-local-qa-only",
        rule_source_version="v1",
        rule_source_hash="a" * 64,
        source_observed_at=observed_at,
        source_available_at=authored_at,
        effective_from=observed_at,
        valid_until=valid_until,
    )
    rule_v2 = ExecutionSubmissionRuleV2(
        execution_rule_v1=rule_v1,
        price_submission_range=PriceSubmissionRangeRule(
            market="SH",
            board="MAIN",
            price_tick=Decimal("0.01"),
            buy_deviation_ratio=Decimal("0.02"),
            sell_deviation_ratio=Decimal("0.02"),
            buy_widening_ticks=10,
            sell_widening_ticks=10,
            minimum_distance_ticks=1,
            lower_floor_ticks=1,
            rounding_mode=PriceRoundingMode.HALF_UP,
        ),
        allowed_stages=(PriceSubmissionStage.CONTINUOUS,),
    )
    settlement_policy = SettlementPolicyV1(
        policy_id=f"ash02-local-settlement-{uuid4().hex}",
        policy_version="v1",
        security_id=security_id,
        market="SH",
        board="MAIN",
        settlement_basis=SettlementBasis.CN_OPEN_MARKET_SESSIONS,
        settlement_lag_sessions=1,
        bound_execution_rule_v2_hash=rule_v2.content_hash,
        source_id="ash02-local-qa-only",
        source_version="v1",
        source_hash="b" * 64,
        source_observed_at=observed_at,
        source_available_at=authored_at,
        effective_from=observed_at,
        valid_until=valid_until,
    )
    bundle = ExecutionSettlementBundleV3(
        bundle_id=f"ash02-local-bundle-{uuid4().hex}",
        bundle_version="v1",
        execution_submission_v2=rule_v2,
        settlement_policy_v1=settlement_policy,
    )
    imported = DjangoAShareExecutionRuleRepository(using="default").import_settlement_bundles(
        bundles=(bundle,)
    )
    assert imported.inserted_count == 1

    cases = tuple(
        FeeTransactionCase(
            asset_type="equity",
            market="SH",
            board="MAIN",
            side=side,
            commission=FeeApplicability.APPLIES,
            transfer_fee=FeeApplicability.APPLIES,
            stamp_duty=(
                FeeApplicability.APPLIES
                if side is SimulatedTradeSide.SELL
                else FeeApplicability.NOT_APPLICABLE
            ),
            slippage=FeeApplicability.APPLIES,
        )
        for side in (SimulatedTradeSide.BUY, SimulatedTradeSide.SELL)
    )
    fee_command = FeePolicyVersionCommand(
        config_name=f"ash02-local-fee-{uuid4().hex}",
        asset_type="all",
        commission_rate_buy=0.0001,
        commission_rate_sell=0.0001,
        min_commission=0.0,
        stamp_duty_rate=0.001,
        transfer_fee_rate=0.00001,
        min_transfer_fee=0.0,
        slippage_rate=0.001,
        policy_id=f"ash02-local-fee-policy-{uuid4().hex}",
        policy_version=1,
        account_scope=FeePolicyAccountScope.ALL_SIMULATED_ACCOUNTS,
        effective_from=observed_at,
        valid_until=valid_until,
        transaction_cases=cases,
        description="private isolated ASH02 acceptance terms; no production authority",
    )
    fee_policy = build_create_fee_policy_version_use_case(using="default").execute(fee_command)
    assert fee_policy.policy_id == fee_command.policy_id


def _post_trade(
    client: APIClient,
    *,
    account_id: int,
    asset_name: str,
    action: str,
    quantity: int,
    key: str,
) -> object:
    """Submit one ordinary authenticated manual-trade API request."""

    return client.post(
        f"/api/simulated-trading/accounts/{account_id}/trade/",
        {
            "asset_code": _ASSET_CODE,
            "asset_name": asset_name,
            "asset_type": "equity",
            "action": action,
            "quantity": quantity,
            "price": "10.00",
            "reason": "private local ASH02 actual-source acceptance",
        },
        format="json",
        HTTP_IDEMPOTENCY_KEY=key,
    )


@pytest.mark.urls("ash02_actual_caller_urls")
def test_exact_actual_sources_reach_normal_manual_trade_and_t_plus_one_rules(
    monkeypatch: pytest.MonkeyPatch,
    record_property: Callable[[str, object], None],
) -> None:
    """Use true captured bytes and production readers to exercise both normal API sides."""

    wall_at_start = timezone.now().astimezone(UTC)
    if wall_at_start.astimezone(ZoneInfo("Asia/Shanghai")).date() != _CAPTURE_START:
        pytest.skip("bounded actual-source candidate is for Sunday 2026-10-18 only")
    if wall_at_start >= _MONDAY_SCENARIO:
        pytest.skip("fixed controlled Monday cutoff passed; recapture source evidence first")

    capture_dir_value = os.environ.get("ASH02_ACTUAL_CAPTURE_DIR", "").strip()
    assert capture_dir_value, "runner must set ASH02_ACTUAL_CAPTURE_DIR to the retained capture folder"
    capture_dir = Path(capture_dir_value).resolve()
    raw_repository = RawLandingRepository()
    responses = load_actual_source_responses(raw_repository, capture_dir=capture_dir)
    replay_client = make_offline_replay_client(responses)

    contract, calendar_policy = _create_calendar_catalog()
    instrument_snapshot = _save_instrument_snapshot(
        repository=raw_repository,
        replay_client=replay_client,
    )
    calendar_snapshot = _save_calendar_publication(
        repository=raw_repository,
        replay_client=replay_client,
        contract=contract,
        policy=calendar_policy,
    )
    assert RawPayloadModel.objects.count() == 2
    assert calendar_snapshot.candidate.raw_reference.response_completed_at == (
        responses.trade_cal.reference.response_completed_at
    )
    assert instrument_snapshot.content.last_response_completed_at == (
        responses.stock_basic.reference.response_completed_at
    )

    source_record = next(
        record
        for fetch_round in instrument_snapshot.content.fetch_rounds
        for record in fetch_round.candidate.records
        if record.ts_code == _ASSET_SOURCE_CODE
    )
    security_id = f"ASH02.{uuid4().hex}"
    runtime_environment = _runtime_environment()
    configure_critical_runtime()
    selector = {
        "schema": "normal-simulated-current-reference-selection.v1",
        "purpose": "normal_simulated_current_reference",
        "max_local_reference_age_seconds": _SCENARIO_TTL_SECONDS,
        "items": [
            {
                "security_id": security_id,
                "asset_code": _ASSET_CODE,
                "source_code": _ASSET_SOURCE_CODE,
                "snapshot_id": str(instrument_snapshot.snapshot_id),
                "snapshot_content_hash": instrument_snapshot.content_hash,
                "snapshot_record_digest": instrument_snapshot.record_digest,
                "mapping_record_id": str(instrument_snapshot.mapping_record.mapping_record_id),
                "mapping_id": instrument_snapshot.mapping_record.content.mapping_id,
                "mapping_version": instrument_snapshot.mapping_record.content.mapping_version,
                "mapping_content_hash": instrument_snapshot.mapping_record.content_hash,
                "mapping_record_digest": instrument_snapshot.mapping_record.record_digest,
            }
        ],
    }
    activate_runtime_profile_patch(
        environment=runtime_environment,
        patch={
            "config_center.market.simulated_current_reference_selection": selector,
            "config_center.market.exchange_timezone_map": {"SSE": "Asia/Shanghai"},
        },
        actor="ash02-local-acceptance",
        reason="pin exact actual-source receipts for isolated local acceptance",
    )

    sunday_cutoff = timezone.now().astimezone(UTC)
    assert sunday_cutoff < _MONDAY_SCENARIO
    assert sunday_cutoff.astimezone(ZoneInfo("Asia/Shanghai")).date() == _CAPTURE_START
    sunday_identity = build_current_reference_identity_reader(
        runtime_environment=runtime_environment
    ).resolve_current_reference_identity(asset_code=_ASSET_CODE, as_of=sunday_cutoff)
    assert sunday_identity is not None
    assert sunday_identity.source_evidence.source_observed_at is None
    assert sunday_identity.source_evidence.source_available_at is None
    assert sunday_identity.source_evidence.must_not_use_for_decision is True
    assert sunday_identity.source_evidence.must_not_execute is True

    scenario_identity = build_current_reference_identity_reader(
        runtime_environment=runtime_environment
    ).resolve_current_reference_identity(
        asset_code=_ASSET_CODE,
        as_of=_MONDAY_SCENARIO,
    )
    assert scenario_identity is not None
    assert scenario_identity.security_id == security_id
    assert scenario_identity.source_evidence.source_observed_at is None
    assert scenario_identity.source_evidence.source_available_at is None
    assert scenario_identity.source_evidence.must_not_use_for_decision is True
    assert scenario_identity.source_evidence.must_not_execute is True

    calendar_reader = build_current_published_execution_calendar_reader(
        dataset_key=contract.dataset_key,
        runtime_environment=runtime_environment,
    )
    sunday_evidence = calendar_reader.get_effective_execution_calendar(
        exchange="SSE",
        as_of=sunday_cutoff,
        coverage_start=None,
    )
    assert sunday_evidence is not None
    with pytest.raises(ValueError, match="explicit open session"):
        sunday_evidence.current_open_session(sunday_cutoff)

    authored_at = timezone.now().astimezone(UTC)
    assert authored_at < _MONDAY_SCENARIO
    _import_local_rule_and_fee(security_id=security_id, authored_at=authored_at)

    user_model = get_user_model()
    user = user_model.objects.create_user(
        username=f"ash02-manual-owner-{uuid4().hex}",
        password="unused-local-test-password",
    )
    account = SimulatedAccountModel.objects.create(
        user=user,
