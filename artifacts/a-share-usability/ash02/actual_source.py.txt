"""Offline replay helpers for the two exact ASH02 source captures.

The capture artifacts were originally retained by
``RetainProviderResponsePayloadUseCase``. This module reconstructs those exact
immutable domain rows, verifies their reference and file digests, and idempotently
lands them through the real ``RawLandingRepository``. It never performs HTTP,
creates a new provider response capture, or substitutes a local timestamp.
"""

from __future__ import annotations

import hashlib
import json
from collections.abc import Mapping
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
from types import MappingProxyType
from typing import Any, Literal, cast

from apps.data_center.application.provider_response_payload import (
    ProviderResponsePayloadReference,
)
from apps.data_center.domain.raw_landing import RawPayload
from apps.data_center.infrastructure.raw_landing_repositories import RawLandingRepository
from apps.data_center.infrastructure.tushare_client import TushareResponseEvidence

DatasetKey = Literal["tushare.stock_basic", "tushare.trade_cal"]
ApiName = Literal["stock_basic", "trade_cal"]

_CAPTURE_DIR = Path(__file__).resolve().parents[1] / "ash02-existing-source-capture-02"
_CAPTURED_DATASETS: tuple[tuple[DatasetKey, ApiName], ...] = (
    ("tushare.stock_basic", "stock_basic"),
    ("tushare.trade_cal", "trade_cal"),
)


class ActualSourceFixtureError(ValueError):
    """Raised when an offline source artifact is absent or no longer exact."""


@dataclass(frozen=True, slots=True)
class CapturedProviderResponse:
    """One exact saved RawPayload and its original provider-response reference."""

    api_name: ApiName
    dataset_key: DatasetKey
    fields: str
    request_parameters: Mapping[str, str]
    row_count: int
    raw_payload: RawPayload
    reference: ProviderResponsePayloadReference

    @property
    def evidence(self) -> TushareResponseEvidence:
        """Build transport evidence from original capture clocks and digests."""

        return TushareResponseEvidence(
            body_sha256=self.reference.body_sha256,
            response_completed_at=self.reference.response_completed_at,
            raw_payload_reference=self.reference,
        )


@dataclass(frozen=True, slots=True)
class ActualSourceResponses:
    """Named access to the exact captured stock-basic and calendar responses."""

    stock_basic: CapturedProviderResponse
    trade_cal: CapturedProviderResponse

    @property
    def by_dataset_key(self) -> Mapping[str, CapturedProviderResponse]:
        """Expose captured responses under their exact immutable dataset keys."""

        return MappingProxyType(
            {
                self.stock_basic.dataset_key: self.stock_basic,
                self.trade_cal.dataset_key: self.trade_cal,
            }
        )


@dataclass(frozen=True, slots=True)
class OfflineTushareReplayResponse:
    """Evidence-only response carrier consumed by production source adapters."""

    response_evidence: TushareResponseEvidence


class OfflineTushareReplayClient:
    """Return only the original evidence for the exact captured request scopes."""

    def __init__(self, responses: ActualSourceResponses) -> None:
        """Bind the verified responses without creating a transport or a clock."""

        self._responses = responses.by_dataset_key

    def stock_basic(
        self,
        *,
        exchange: str,
        list_status: str,
        fields: str,
    ) -> OfflineTushareReplayResponse:
        """Replay the captured SSE/L stock-basic response for its exact fields."""

        return self._replay(
            "tushare.stock_basic",
            {"exchange": exchange, "list_status": list_status, "fields": fields},
        )

    def trade_cal(
        self,
        *,
        exchange: str,
        start_date: str,
        end_date: str,
        fields: str,
    ) -> OfflineTushareReplayResponse:
        """Replay the captured SSE calendar response for its exact nine-day range."""

        return self._replay(
            "tushare.trade_cal",
            {
                "exchange": exchange,
                "start_date": start_date,
                "end_date": end_date,
                "fields": fields,
            },
        )

    def _replay(
        self,
        dataset_key: DatasetKey,
        request_parameters: Mapping[str, str],
    ) -> OfflineTushareReplayResponse:
        """Require exact captured scope and return its existing evidence only."""

        captured = self._responses.get(dataset_key)
        if captured is None or dict(request_parameters) != dict(captured.request_parameters):
            raise ActualSourceFixtureError("offline_replay_request_scope_mismatch")
        return OfflineTushareReplayResponse(response_evidence=captured.evidence)


def load_actual_source_responses(
    repository: RawLandingRepository,
    capture_dir: Path | None = None,
) -> ActualSourceResponses:
    """Verify and idempotently save both exact capture rows through RawLandingRepository.

    ``fetched_at`` and ``response_completed_at`` come from the original capture
    files and are never normalized or replaced. The repository must use the
    default database alias because the production source and audit adapters use
    that same evidence boundary.
    """

    if repository.unit_of_work_key != "django:default":
        raise ActualSourceFixtureError("raw_landing_repository_must_use_default_alias")
    root = capture_dir or _CAPTURE_DIR
    root = root.resolve()
    report = _read_json_object(root / "report.json")
    attempts = _index_rows(report.get("attempts"), "api_name")
    raw_records = _index_rows(report.get("raw_payload_records"), "dataset_key")
    loaded: dict[DatasetKey, CapturedProviderResponse] = {}

    if set(attempts) != {api_name for _dataset_key, api_name in _CAPTURED_DATASETS}:
        raise ActualSourceFixtureError("capture_attempt_scope_mismatch")
    if set(raw_records) != {dataset_key for dataset_key, _api_name in _CAPTURED_DATASETS}:
        raise ActualSourceFixtureError("capture_raw_payload_scope_mismatch")

    for dataset_key, api_name in _CAPTURED_DATASETS:
        attempt = attempts[api_name]
        raw_record = raw_records[dataset_key]
        if attempt.get("status") != "captured":
            raise ActualSourceFixtureError("capture_attempt_not_successful")
        reference_payload = _object(attempt.get("reference"), "attempt.reference")
        file_name = _text(raw_record.get("file"), "raw_payload_records.file")
        if Path(file_name).name != file_name or not file_name.endswith(".raw-payload.json"):
            raise ActualSourceFixtureError("capture_payload_filename_invalid")
        payload_path = (root / file_name).resolve()
        if payload_path.parent != root:
            raise ActualSourceFixtureError("capture_payload_path_invalid")
        encoded_payload = payload_path.read_bytes()
        if hashlib.sha256(encoded_payload).hexdigest() != _text(
            raw_record.get("raw_sha256"), "raw_payload_records.raw_sha256"
        ):
            raise ActualSourceFixtureError("capture_payload_file_digest_mismatch")
        outer = _decode_object(encoded_payload, "RawPayload file")
        raw_payload = _raw_payload_from_object(outer)
        reference = ProviderResponsePayloadReference.from_payload(
            cast(Mapping[str, object], reference_payload)
        )
        _validate_pair(
            dataset_key=dataset_key,
            api_name=api_name,
            attempt=attempt,
            raw_record=raw_record,
            raw_payload=raw_payload,
            reference=reference,
        )
        request_parameters = _request_parameters(raw_payload)
        fields = request_parameters.get("fields")
        if fields is None:
            raise ActualSourceFixtureError("capture_fields_missing")
        row_count = _reported_row_count(attempt)
        if _retained_response_row_count(raw_payload) != row_count:
            raise ActualSourceFixtureError("capture_row_count_mismatch")

        stored = repository.save(raw_payload)
        if stored != raw_payload:
            raise ActualSourceFixtureError("raw_payload_repository_changed_capture")
        read_back = repository.get_by_hash(reference.payload_hash)
        if (
            read_back is None
            or read_back != raw_payload
            or not reference.verifies(read_back)
        ):
            raise ActualSourceFixtureError("raw_payload_repository_readback_mismatch")

        loaded[dataset_key] = CapturedProviderResponse(
            api_name=api_name,
            dataset_key=dataset_key,
            fields=fields,
            request_parameters=MappingProxyType(request_parameters),
            row_count=row_count,
            raw_payload=read_back,
            reference=reference,
        )

    return ActualSourceResponses(
        stock_basic=loaded["tushare.stock_basic"],
        trade_cal=loaded["tushare.trade_cal"],
    )


def make_offline_replay_client(
    responses: ActualSourceResponses,
) -> OfflineTushareReplayClient:
    """Create the offline evidence client consumed by production source adapters."""

    return OfflineTushareReplayClient(responses)


def _read_json_object(path: Path) -> dict[str, object]:
    """Read one UTF-8 JSON object without importing project code or network clients."""

    try:
        return _decode_object(path.read_bytes(), str(path))
    except OSError as exc:
        raise ActualSourceFixtureError("capture_artifact_missing") from exc


def _decode_object(encoded: bytes, source: str) -> dict[str, object]:
    """Decode one JSON object and reject malformed top-level shapes."""

    try:
        value = json.loads(encoded.decode("utf-8"))
    except (UnicodeDecodeError, json.JSONDecodeError) as exc:
        raise ActualSourceFixtureError(f"{source}_invalid_json") from exc
    return _object(value, source)


def _object(value: object, name: str) -> dict[str, object]:
    """Narrow a JSON boundary to a string-keyed object."""

    if not isinstance(value, dict) or any(not isinstance(key, str) for key in value):
        raise ActualSourceFixtureError(f"{name}_must_be_object")
    return cast(dict[str, object], value)


def _index_rows(value: object, key: str) -> dict[str, dict[str, object]]:
    """Index a finite manifest list by a required unique string key."""

    if not isinstance(value, list):
        raise ActualSourceFixtureError("capture_manifest_rows_invalid")
    result: dict[str, dict[str, object]] = {}
    for raw_row in value:
        row = _object(raw_row, "capture manifest row")
        identifier = _text(row.get(key), key)
        if identifier in result:
            raise ActualSourceFixtureError("capture_manifest_duplicate_key")
        result[identifier] = row
    return result


def _text(value: object, name: str) -> str:
    """Require one non-empty string from a JSON boundary."""

    if not isinstance(value, str) or not value:
        raise ActualSourceFixtureError(f"{name}_must_be_nonempty_text")
    return value


def _datetime(value: object, name: str) -> datetime:
    """Parse an aware source timestamp without clock or timezone substitution."""

    try:
        timestamp = datetime.fromisoformat(_text(value, name))
    except ValueError as exc:
        raise ActualSourceFixtureError(f"{name}_invalid_datetime") from exc
    if timestamp.tzinfo is None or timestamp.utcoffset() is None:
        raise ActualSourceFixtureError(f"{name}_must_be_aware")
    return timestamp


def _raw_payload_from_object(value: dict[str, object]) -> RawPayload:
    """Rebuild exactly the immutable RawPayload metadata from its saved artifact."""

    payload = _object(value.get("payload"), "RawPayload.payload")
    request_params = _object(value.get("request_params"), "RawPayload.request_params")
    retention_raw = value.get("retention_until")
    retention_until = (
        _datetime(retention_raw, "retention_until") if retention_raw is not None else None
    )
    return RawPayload(
        payload_id=_text(value.get("payload_id"), "payload_id"),
        dataset_key=_text(value.get("dataset_key"), "dataset_key"),
        provider_name=_text(value.get("provider_name"), "provider_name"),
        payload_hash=_text(value.get("payload_hash"), "payload_hash"),
        schema_fingerprint=_text(value.get("schema_fingerprint"), "schema_fingerprint"),
        payload=cast(dict[str, Any], payload),
        fetched_at=_datetime(value.get("fetched_at"), "fetched_at"),
        request_params=cast(dict[str, Any], request_params),
        run_id=_text(value.get("run_id"), "run_id") if value.get("run_id") else "",
        batch_id=_text(value.get("batch_id"), "batch_id") if value.get("batch_id") else "",
        content_type=_text(value.get("content_type"), "content_type"),
        parser_version=_text(value.get("parser_version"), "parser_version"),
        redacted=value.get("redacted") is True,
        payload_size_bytes=_nonnegative_int(
            value.get("payload_size_bytes"), "payload_size_bytes"
        ),
        retention_until=retention_until,
    )


def _request_parameters(payload: RawPayload) -> dict[str, str]:
    """Flatten only the original endpoint parameters plus requested fields."""

    request = _object(payload.request_params, "RawPayload.request_params")
    params = _object(request.get("params"), "RawPayload.request_params.params")
    fields = _text(request.get("fields"), "RawPayload.request_params.fields")
    flattened = {"fields": fields}
    for key, value in params.items():
        flattened[key] = _text(value, f"request_params.{key}")
    return flattened


def _validate_pair(
    *,
    dataset_key: DatasetKey,
    api_name: ApiName,
    attempt: Mapping[str, object],
    raw_record: Mapping[str, object],
    raw_payload: RawPayload,
    reference: ProviderResponsePayloadReference,
) -> None:
    """Bind the manifest, original file, payload, response body and reference."""

    if (
        raw_payload.dataset_key != dataset_key
        or raw_payload.provider_name != "tushare"
        or raw_record.get("dataset_key") != dataset_key
        or attempt.get("dataset_key") != dataset_key
        or attempt.get("api_name") != api_name
        or raw_record.get("payload_id") != raw_payload.payload_id
        or raw_record.get("payload_hash") != raw_payload.payload_hash
        or raw_record.get("fetched_at") != raw_payload.fetched_at.isoformat()
        or reference.provider_name != raw_payload.provider_name
        or reference.dataset_key != dataset_key
        or reference.payload_id != raw_payload.payload_id
        or reference.payload_hash != raw_payload.payload_hash
        or reference.response_completed_at != raw_payload.fetched_at
        or reference.original_body_retained is not True
        or reference.retained_content_encoding != "exact_original_utf8_json"
        or not reference.verifies(raw_payload)
    ):
        raise ActualSourceFixtureError("capture_reference_binding_mismatch")

    body = raw_payload.payload.get("retained_response_json")
    if not isinstance(body, str):
        raise ActualSourceFixtureError("capture_original_body_missing")
    body_bytes = body.encode("utf-8")
    if (
        hashlib.sha256(body_bytes).hexdigest() != reference.body_sha256
        or len(body_bytes) != reference.body_size_bytes
        or raw_payload.payload.get("transport") != reference.transport.value
    ):
        raise ActualSourceFixtureError("capture_original_body_digest_mismatch")

    manifest_parameters = _object(attempt.get("parameters"), "attempt.parameters")
    if manifest_parameters != _request_parameters(raw_payload):
        raise ActualSourceFixtureError("capture_request_parameter_mismatch")


def _reported_row_count(attempt: Mapping[str, object]) -> int:
    """Require a non-negative exact row count from the capture manifest."""

    return _nonnegative_int(attempt.get("rows"), "attempt.rows")


def _retained_response_row_count(payload: RawPayload) -> int:
    """Count the original response items without normalizing their source rows."""

    body = payload.payload.get("retained_response_json")
    if not isinstance(body, str):
        raise ActualSourceFixtureError("capture_original_body_missing")
    try:
        response = _object(json.loads(body), "retained_response_json")
        data = _object(response.get("data"), "retained_response_json.data")
        items = data.get("items")
    except json.JSONDecodeError as exc:
        raise ActualSourceFixtureError("capture_original_body_invalid_json") from exc
    if not isinstance(items, list):
        raise ActualSourceFixtureError("capture_original_body_items_invalid")
    return len(items)


def _nonnegative_int(value: object, name: str) -> int:
    """Require a non-negative integer and exclude boolean JSON values."""

    if isinstance(value, bool) or not isinstance(value, int) or value < 0:
        raise ActualSourceFixtureError(f"{name}_must_be_nonnegative_integer")
    return value
