"""Typed, fail-closed selection of one configured exchange-calendar source."""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from typing import Protocol

from apps.data_center.domain.contracts import (
    DatasetContract,
    DatasetKey,
    ProviderBinding,
    PublicationPolicy,
)
from apps.data_center.domain.control_plane import CalendarNullableMemberCodecVersion
from core.exceptions import DataFetchError

CALENDAR_CAPTURE_CAPABILITY = "exchange_calendar"
CALENDAR_CAPTURE_PROVIDER = "tushare"
CALENDAR_CAPTURE_VALIDATOR_KEY = "exchange_calendar.tushare.trade_cal.full_date.v1"


class CalendarCaptureConfigurationError(DataFetchError):
    """A missing, ambiguous, stale, or unsupported capture configuration."""


@dataclass(frozen=True, slots=True)
class CalendarCaptureContractCandidate:
    """One active Dataset Contract eligible to configure calendar capture."""

    contract: DatasetContract
    active: bool
    created_at: datetime
    updated_at: datetime

    def __post_init__(self) -> None:
        """Revalidate the persisted contract and its knowledge timestamps."""

        if type(self.contract) is not DatasetContract or type(self.active) is not bool:
            raise TypeError("Calendar capture contract candidate is invalid")
        self.contract.__post_init__()
        _require_utc(self.created_at, "contract.created_at")
        _require_utc(self.updated_at, "contract.updated_at")


@dataclass(frozen=True, slots=True)
class CalendarCapturePolicyCandidate:
    """One active publication policy and its persisted knowledge window."""

    policy: PublicationPolicy
    active: bool
    created_at: datetime
    updated_at: datetime

    def __post_init__(self) -> None:
        """Revalidate the policy before configuration selection."""

        if type(self.policy) is not PublicationPolicy or type(self.active) is not bool:
            raise TypeError("Calendar capture policy candidate is invalid")
        self.policy.__post_init__()
        _require_utc(self.created_at, "policy.created_at")
        _require_utc(self.updated_at, "policy.updated_at")


@dataclass(frozen=True, slots=True)
class CalendarCaptureBindingCandidate:
    """One enabled provider binding and its persisted knowledge window."""

    binding: ProviderBinding
    enabled: bool
    created_at: datetime
    updated_at: datetime

    def __post_init__(self) -> None:
        """Revalidate the binding and ensure its state flag is exact."""

        if type(self.binding) is not ProviderBinding or type(self.enabled) is not bool:
            raise TypeError("Calendar capture binding candidate is invalid")
        self.binding.__post_init__()
        _require_utc(self.created_at, "binding.created_at")
        _require_utc(self.updated_at, "binding.updated_at")


@dataclass(frozen=True, slots=True)
class CalendarCaptureProviderCandidate:
    """Non-secret metadata for one active provider instance."""

    provider_id: int
    source_type: str
    is_active: bool
    created_at: datetime
    updated_at: datetime

    def __post_init__(self) -> None:
        """Validate identity, provider type, state, and metadata timestamps."""

        if type(self.provider_id) is not int or self.provider_id < 1:
            raise ValueError("Calendar capture provider ID is invalid")
        if type(self.source_type) is not str or not self.source_type:
            raise ValueError("Calendar capture provider source type is invalid")
        if type(self.is_active) is not bool:
            raise TypeError("Calendar capture provider state must be a boolean")
        _require_utc(self.created_at, "provider.created_at")
        _require_utc(self.updated_at, "provider.updated_at")


@dataclass(frozen=True, slots=True)
class CalendarCaptureConfigurationCandidates:
    """All eligible catalog and provider rows observed inside one DB fence."""

    contracts: tuple[CalendarCaptureContractCandidate, ...]
    policies: tuple[CalendarCapturePolicyCandidate, ...]
    bindings: tuple[CalendarCaptureBindingCandidate, ...]
    providers: tuple[CalendarCaptureProviderCandidate, ...]

    def __post_init__(self) -> None:
        """Reject mutable collections and malformed rows at the repository edge."""

        for field_name, entries, entry_type in (
            ("contracts", self.contracts, CalendarCaptureContractCandidate),
            ("policies", self.policies, CalendarCapturePolicyCandidate),
            ("bindings", self.bindings, CalendarCaptureBindingCandidate),
            ("providers", self.providers, CalendarCaptureProviderCandidate),
        ):
            if type(entries) is not tuple:
                raise TypeError(f"Calendar capture {field_name} must be a tuple")
            for entry in entries:
                if type(entry) is not entry_type:
                    raise TypeError(f"Calendar capture {field_name} contains an invalid row")
                entry.__post_init__()


@dataclass(frozen=True, slots=True)
class CalendarCaptureConfiguration:
    """Exact source selection used to build one capture operation."""

    dataset: DatasetKey
    calendar_response_max_age_seconds: int
    policy: PublicationPolicy
    binding: ProviderBinding
    provider_id: int
    validator_key: str
    known_at: datetime

    def __post_init__(self) -> None:
        """Recheck exact version bindings before infrastructure uses the selection."""

        if type(self.dataset) is not DatasetKey:
            raise TypeError("Calendar capture dataset key is invalid")
        self.dataset.__post_init__()
        if type(self.calendar_response_max_age_seconds) is not int or (
            self.calendar_response_max_age_seconds < 1
        ):
            raise ValueError("Calendar capture response TTL is invalid")
        if type(self.policy) is not PublicationPolicy or self.policy.dataset != self.dataset:
            raise ValueError("Calendar capture publication policy version differs")
        self.policy.__post_init__()
        if self.policy.member_codec_version is not CalendarNullableMemberCodecVersion.V1:
            raise ValueError("Calendar capture publication codec is unsupported")
        if type(self.binding) is not ProviderBinding or self.binding.dataset != self.dataset:
            raise ValueError("Calendar capture provider binding version differs")
        self.binding.__post_init__()
        if self.binding.provider != CALENDAR_CAPTURE_PROVIDER:
            raise ValueError("Calendar capture provider is unsupported")
        if self.binding.capability != CALENDAR_CAPTURE_CAPABILITY:
            raise ValueError("Calendar capture capability is unsupported")
        if self.binding.validator_key != CALENDAR_CAPTURE_VALIDATOR_KEY:
            raise ValueError("Calendar capture validator is unsupported")
        if type(self.provider_id) is not int or self.provider_id < 1:
            raise ValueError("Calendar capture provider ID is invalid")
        if type(self.validator_key) is not str or self.validator_key != (
            CALENDAR_CAPTURE_VALIDATOR_KEY
        ):
            raise ValueError("Calendar capture validator key is invalid")
        _require_utc(self.known_at, "known_at")


class CalendarCaptureConfigurationCandidatesPort(Protocol):
    """Read eligible configuration rows under the caller-owned DB transaction."""

    def load_candidates(
        self,
        *,
        using: str,
    ) -> CalendarCaptureConfigurationCandidates:
        """Return rows protected by a same-alias metadata fence."""


class SelectCalendarCaptureConfigurationUseCase:
    """Select exactly one source configuration at an explicit knowledge time."""

    def __init__(self, repository: CalendarCaptureConfigurationCandidatesPort) -> None:
        """Bind a read-only catalog candidate repository."""

        self._repository = repository

    def execute(
        self,
        *,
        known_at: datetime,
        using: str = "default",
    ) -> CalendarCaptureConfiguration:
        """Resolve current active config, without reconstructing historical catalog state."""

        _validate_knowledge_time(known_at)
        candidates = self._repository.load_candidates(using=using)
        return resolve_calendar_capture_configuration(candidates, known_at=known_at)


def resolve_calendar_capture_configuration(
    candidates: CalendarCaptureConfigurationCandidates,
    *,
    known_at: datetime,
) -> CalendarCaptureConfiguration:
    """Resolve exact active rows and reject metadata learned after ``known_at``.

    This only validates currently active configuration. The catalog does not
    retain enough history to reconstruct which configuration was active at an
    earlier cutoff.
    """

    if type(candidates) is not CalendarCaptureConfigurationCandidates:
        raise _blocked("CALENDAR_CAPTURE_CANDIDATES_INVALID")
    try:
        _validate_knowledge_time(known_at)
    except CalendarCaptureConfigurationError:
        raise
    try:
        candidates.__post_init__()
    except (AttributeError, TypeError, ValueError) as error:
        raise _blocked("CALENDAR_CAPTURE_CANDIDATES_INVALID") from error

    contracts = tuple(
        candidate
        for candidate in candidates.contracts
        if candidate.active and candidate.contract.calendar_response_max_age_seconds is not None
    )
    if not contracts:
        raise _blocked("CALENDAR_CAPTURE_DATASET_MISSING")
    if len(contracts) != 1:
        raise _blocked("CALENDAR_CAPTURE_DATASET_AMBIGUOUS")
    contract_candidate = contracts[0]
    contract = contract_candidate.contract
    dataset = contract.key
    ttl = contract.calendar_response_max_age_seconds
    if type(ttl) is not int or ttl < 1:
        raise _blocked("CALENDAR_CAPTURE_TTL_MISSING")

    same_dataset_policies = tuple(
        candidate
        for candidate in candidates.policies
        if candidate.active and candidate.policy.dataset.value == dataset.value
    )
    if not same_dataset_policies:
        raise _blocked("CALENDAR_CAPTURE_POLICY_MISSING")
    if len(same_dataset_policies) != 1:
        raise _blocked("CALENDAR_CAPTURE_POLICY_AMBIGUOUS")
    policy_candidate = same_dataset_policies[0]
    policy = policy_candidate.policy
    if policy.dataset != dataset:
        raise _blocked("CALENDAR_CAPTURE_POLICY_VERSION_MISMATCH")
    if policy.member_codec_version is not CalendarNullableMemberCodecVersion.V1:
        raise _blocked("CALENDAR_CAPTURE_POLICY_CODEC_UNSUPPORTED")

    same_dataset_bindings = tuple(
        candidate
        for candidate in candidates.bindings
        if candidate.enabled
        and candidate.binding.capability == CALENDAR_CAPTURE_CAPABILITY
        and candidate.binding.dataset.value == dataset.value
    )
    if not same_dataset_bindings:
        raise _blocked("CALENDAR_CAPTURE_BINDING_MISSING")
    if len(same_dataset_bindings) != 1:
        raise _blocked("CALENDAR_CAPTURE_BINDING_AMBIGUOUS")
    binding_candidate = same_dataset_bindings[0]
    binding = binding_candidate.binding
    if binding.dataset != dataset:
        raise _blocked("CALENDAR_CAPTURE_BINDING_VERSION_MISMATCH")
    if binding.provider != CALENDAR_CAPTURE_PROVIDER:
        raise _blocked("CALENDAR_CAPTURE_PROVIDER_UNSUPPORTED")
    if binding.validator_key != CALENDAR_CAPTURE_VALIDATOR_KEY:
        raise _blocked("CALENDAR_CAPTURE_VALIDATOR_UNSUPPORTED")

    providers = tuple(
        candidate
        for candidate in candidates.providers
        if candidate.is_active and candidate.source_type == CALENDAR_CAPTURE_PROVIDER
    )
    if not providers:
        raise _blocked("CALENDAR_CAPTURE_PROVIDER_INSTANCE_MISSING")
    if len(providers) != 1:
        raise _blocked("CALENDAR_CAPTURE_PROVIDER_INSTANCE_AMBIGUOUS")
    provider_candidate = providers[0]

    selected_rows = (
        (contract_candidate.created_at, contract_candidate.updated_at),
        (policy_candidate.created_at, policy_candidate.updated_at),
        (binding_candidate.created_at, binding_candidate.updated_at),
        (provider_candidate.created_at, provider_candidate.updated_at),
    )
    if any(
        created_at > known_at or updated_at > known_at for created_at, updated_at in selected_rows
    ):
        raise _blocked("CALENDAR_CAPTURE_METADATA_NOT_KNOWN")

    return CalendarCaptureConfiguration(
        dataset=dataset,
        calendar_response_max_age_seconds=ttl,
        policy=policy,
        binding=binding,
        provider_id=provider_candidate.provider_id,
        validator_key=binding.validator_key,
        known_at=known_at,
    )


def _blocked(code: str) -> CalendarCaptureConfigurationError:
    """Build one stable, credential-free configuration error."""

    return CalendarCaptureConfigurationError(
        "Exchange calendar capture configuration is missing, ambiguous, stale, or unsupported",
        code=code,
    )


def _validate_knowledge_time(value: object) -> None:
    """Require an explicit UTC configuration cutoff before any repository read."""

    try:
        _require_utc(value, "known_at")
    except (TypeError, ValueError) as error:
        raise _blocked("CALENDAR_CAPTURE_KNOWLEDGE_TIME_INVALID") from error


def _require_utc(value: object, field_name: str) -> None:
    """Require an aware zero-offset timestamp for persisted metadata."""

    if not isinstance(value, datetime) or not _is_utc(value):
        raise ValueError(f"{field_name} must be a timezone-aware UTC instant")


def _is_utc(value: datetime) -> bool:
    """Return whether one datetime carries a zero UTC offset."""

    offset = value.utcoffset()
    return value.tzinfo is not None and offset is not None and offset.total_seconds() == 0
