"""Behavioral regressions for simulated A-share settlement and odd-lot sells."""

from datetime import date

import pytest

from apps.simulated_trading.application.use_cases import (
    ExecuteBuyOrderUseCase,
    ExecuteSellOrderUseCase,
)
from apps.simulated_trading.domain.entities import (
    AccountType,
    FeeConfig,
    Position,
    SimulatedAccount,
    SimulatedTrade,
)


class _AccountRepository:
    def __init__(self, account: SimulatedAccount) -> None:
        self.account = account
        self.saved: list[SimulatedAccount] = []

    def get_by_id(self, account_id: int) -> SimulatedAccount | None:
        return self.account if self.account.account_id == account_id else None

    def save(self, account: SimulatedAccount, user_id: int | None = None) -> int:
        self.account = account
        self.saved.append(account)
        return account.account_id


class _PositionRepository:
    def __init__(self, position: Position | None = None) -> None:
        self.positions: dict[tuple[int, str], Position] = {}
        self.saved: list[Position] = []
        self.deleted: list[tuple[int, str]] = []
        if position is not None:
            self.positions[(position.account_id, position.asset_code)] = position

    def get_position(self, account_id: int, asset_code: str) -> Position | None:
        return self.positions.get((account_id, asset_code))

    def save(self, position: Position) -> int:
        self.positions[(position.account_id, position.asset_code)] = position
        self.saved.append(position)
        return len(self.saved)

    def delete(self, account_id: int, asset_code: str) -> bool:
        self.deleted.append((account_id, asset_code))
        return self.positions.pop((account_id, asset_code), None) is not None


class _TradeRepository:
    def __init__(self) -> None:
        self.saved: list[SimulatedTrade] = []

    def save(self, trade: SimulatedTrade) -> int:
        self.saved.append(trade)
        return len(self.saved)


class _FeeConfigRepository:
    def __init__(self) -> None:
        self.config = FeeConfig(
            config_id=1,
            config_name="ash01-test",
            asset_type="equity",
            commission_rate_buy=0.0,
            commission_rate_sell=0.0,
            min_commission=0.0,
            stamp_duty_rate=0.0,
            transfer_fee_rate=0.0,
            min_transfer_fee=0.0,
            slippage_rate=0.0,
            is_default=True,
        )

    def get_default_config(self, asset_type: str = "all") -> FeeConfig | None:
        return self.config if asset_type == "equity" else None


def _account(*, market_value: float = 0.0) -> SimulatedAccount:
    cash = 100_000.0
    return SimulatedAccount(
        account_id=1,
        account_name="ASH-01 test",
        account_type=AccountType.SIMULATED,
        initial_capital=cash,
        current_cash=cash,
        current_market_value=market_value,
        total_value=cash + market_value,
        max_position_pct=100.0,
    )


def _position(*, quantity: float, available_quantity: float) -> Position:
    price = 10.0
    return Position(
        account_id=1,
        asset_code="600000.SH",
        asset_name="浦发银行",
        asset_type="equity",
        quantity=quantity,
        available_quantity=available_quantity,
        avg_cost=price,
        total_cost=quantity * price,
        current_price=price,
        market_value=quantity * price,
        unrealized_pnl=0.0,
        unrealized_pnl_pct=0.0,
        first_buy_date=date(2026, 10, 15),
        last_update_date=date(2026, 10, 15),
    )


def test_new_a_share_buy_is_not_sellable_on_execution_day() -> None:
    account_repo = _AccountRepository(_account())
    position_repo = _PositionRepository()
    buy = ExecuteBuyOrderUseCase(
        account_repo,
        position_repo,
        _TradeRepository(),
        _FeeConfigRepository(),
    )

    buy.execute(
        account_id=1,
        asset_code="600000.SH",
        asset_name="浦发银行",
        asset_type="equity",
        quantity=100,
        price=10.0,
        execution_date=date(2026, 10, 16),
    )

    position = position_repo.get_position(1, "600000.SH")
    assert position is not None
    assert position.quantity == 100.0
    assert position.available_quantity == 0.0


def test_a_share_addition_keeps_only_prior_day_shares_sellable() -> None:
    existing = _position(quantity=300.0, available_quantity=300.0)
    account_repo = _AccountRepository(_account(market_value=3_000.0))
    position_repo = _PositionRepository(existing)
    buy = ExecuteBuyOrderUseCase(
        account_repo,
        position_repo,
        _TradeRepository(),
        _FeeConfigRepository(),
    )

    buy.execute(
        account_id=1,
        asset_code="600000.SH",
        asset_name="浦发银行",
        asset_type="equity",
        quantity=200,
        price=10.0,
        execution_date=date(2026, 10, 16),
    )

    position = position_repo.get_position(1, "600000.SH")
    assert position is not None
    assert position.quantity == 500.0
    assert position.available_quantity == 300.0


def test_a_share_odd_lot_can_be_sold_when_closing_the_entire_position() -> None:
    position_repo = _PositionRepository(_position(quantity=150.0, available_quantity=150.0))
    trade_repo = _TradeRepository()
    sell = ExecuteSellOrderUseCase(
        _AccountRepository(_account(market_value=1_500.0)),
        position_repo,
        trade_repo,
        _FeeConfigRepository(),
    )

    trade = sell.execute(
        account_id=1,
        asset_code="600000.SH",
        quantity=150,
        price=11.0,
        execution_date=date(2026, 10, 16),
    )

    assert trade.quantity == 150.0
    assert position_repo.get_position(1, "600000.SH") is None
    assert len(trade_repo.saved) == 1


def test_a_share_odd_remainder_can_be_sold_once_without_closing_whole_lots() -> None:
    position_repo = _PositionRepository(_position(quantity=150.0, available_quantity=150.0))
    trade_repo = _TradeRepository()
    sell = ExecuteSellOrderUseCase(
        _AccountRepository(_account(market_value=1_500.0)),
        position_repo,
        trade_repo,
        _FeeConfigRepository(),
    )

    trade = sell.execute(
        account_id=1,
        asset_code="600000.SH",
        quantity=50,
        price=11.0,
        execution_date=date(2026, 10, 16),
    )

    remaining = position_repo.get_position(1, "600000.SH")
    assert trade.quantity == 50.0
    assert remaining is not None
    assert remaining.quantity == 100.0
    assert remaining.available_quantity == 100.0
    assert len(trade_repo.saved) == 1


def test_a_share_odd_remainder_cannot_be_split_into_multiple_odd_lots() -> None:
    existing = _position(quantity=150.0, available_quantity=150.0)
    position_repo = _PositionRepository(existing)
    trade_repo = _TradeRepository()
    sell = ExecuteSellOrderUseCase(
        _AccountRepository(_account(market_value=1_500.0)),
        position_repo,
        trade_repo,
        _FeeConfigRepository(),
    )

    with pytest.raises(ValueError):
        sell.execute(
            account_id=1,
            asset_code="600000.SH",
            quantity=25,
            price=11.0,
            execution_date=date(2026, 10, 16),
        )

    assert position_repo.get_position(1, "600000.SH") is existing
    assert position_repo.saved == []
    assert position_repo.deleted == []
    assert trade_repo.saved == []
